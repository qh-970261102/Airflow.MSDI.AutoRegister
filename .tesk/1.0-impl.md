# 要求
1. 1.0版本仅仅面向最基础的msdi，因此引用的版本是msdi1.0
2. SingletonServiceAttribute、ScopedServiceAttribute、TransientServiceAttribute是最基础的特性，这些就是简单的描述目标类或者目标方法的注册方式，可以看到这三个有自己的metadate特性描述，很清楚的是，这个有index有default，这个metadata特性就是服务用户想新增的自定义特性的，因此对于BaseServiceAttribute的所有子类特性都应该使用RegistrationOptionsAttribute描述
3. 目前目标就是将AutoRegisterAttribute描述的函数自动生成其扩展class实现，这部分将在下边的实例给出
4. 目前需要面向class和function两种方式注册，需要有配套的警告机制


# 示例

```csharp
// 目标service
[SingletonService]
class ServiceA{

}

[ScopedService]
class ServiceB{

}

[TransientService]
class ServiceC{

}

[SingletonService(typeof(InterfaceD))]
class ServiceD: InterfaceD{

}

[SingletonService]
[SingletonService(typeof(InterfaceE))]
class ServiceE: InterfaceE{

}

[SingletonService(typeof(InterfaceF), 2)]
class ServiceF: InterfaceF{

}

[SingletonService(typeof(InterfaceF), 1)]
class ServiceG: InterfaceF{

}

class ServiceH{
    private readonly InterfaceF _serviceG;
    public ServiceH(InterfaceF serviceG){
        this._serviceG = serviceG;
    }
    [SingletonService]
    public ServiceI GetServiceI(){
        return new ServiceI(_serviceG);
    }
    [ScopedService]
    public ServiceJ GetServiceJ(IServiceProvider sp){
        return new ServiceJ(_serviceG);
    }
}


// autoregister

internal static partial class Register{

    [AutoRegister]
    public static partial void AddServicesFromCurrentAssembly(IServiceCollection services);


    public static IServiceCollection AddServices(this IServiceCollection services){
        AddServicesFromCurrentAssembly(services);
        return services;
    }
}


// 自动生成的目标类
internal static partial class Register{
    public static void AddServicesFromCurrentAssembly(IServiceCollection services){
        services.AddSingleton<ServiceA>();
        services.AddScoped<ServiceB>();
        services.AddTransient<ServiceC>();
        services.AddSingleton<InterfaceD, ServiceD>();

        // 同一个类的所有单例注册应该一致的指向同一份
        services.AddSingleton<ServiceE>();
        services.AddSingleton<InterfaceE>(sp => sp.GetRequiredService<ServiceE>());

        // 注意order
        services.AddSingleton<InterfaceF, ServiceG>();
        services.AddSingleton<InterfaceF, ServiceF>();

        // 对于工厂注册，需要先注册单例的其所属类
        // 考虑是否要求类只能是单纯的类，是否可以添加自定义error来硬约束这个，还要硬约束不能在类上再添加注册特性了，因为已经自动注册为单例，还要对函数格式做出要求，只能是无参或者有一个IServiceProvider参数
        // 注意自动适配参数
        // 生成器对 [SingletonService] 且带 IServiceProvider 参数的方法直接报错/警告（单例工厂只允许无参）；[Scoped]/[Transient] 工厂里 resolve 单例没问题。单例工厂方法的依赖应该全部来自持有类的构造注入（持有类本身是单例，它的构造依赖必须是单例，已经在编译期校验过了）。如果这个方法需要解析 scoped 依赖——那说明它根本不是单例，生命周期用错了。如果只需要 singleton 依赖——那走构造注入就够了，根本不需要 sp。
        services.AddSingleton<ServiceH>();
        services.AddSingleton(sp=>sp.GetRequiredService<ServiceH>().GetServiceI());
        services.AddScoped(sp=>sp.GetRequiredService<ServiceH>().GetServiceJ(sp));
    }
}

```
